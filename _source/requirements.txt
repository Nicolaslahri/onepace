# Mozilla's CA bundle. The app trusts this first and falls back to the OS
# certificate store — see tls_trust.py. Without it, users whose Windows root
# store predates ISRG Root X2 can't reach pixeldrain at all.
#
# Pin-bump this on every release: a bundle frozen into a .onefile .exe never
# self-updates, so a stale pin reintroduces exactly the bug it fixes.
certifi>=2026.7.22

# Build with .\build.ps1 — it carries the --collect-data certifi flag and the
# --add-data list for the bundled .json catalogs. Building by hand without
# those produces an .exe that either can't reach pixeldrain or has no data.
pyinstaller>=6.0
